class RoutingNetwork:
    def __init__(self):
        self.graph = nx.Graph()
        self.node_coords = {}
        self.epsg = None

    def add_node(self, node_id, x, y):
        """Add a georeferenced node."""
        self.graph.add_node(node_id)
        self.node_coords[node_id] = (x, y)

    def add_edge(self, u, v, weight, **attrs):
        """Add a weighted edge between two nodes."""
        self.graph.add_edge(u, v, weight=weight, **attrs)

    def polylines_to_graph(trails_vector: Vector, snap_distance=50) -> RoutingNetwork:
        """Convert trail polylines to routing graph with node snapping.

        Args:
            trails_vector: Vector instance with POLYLINE geometry
            snap_distance: Distance in map units to snap endpoint nodes (meters, default 50)

        Returns:
            RoutingNetwork instance with graph topology

        Node IDs: Sequential integers starting from 0 (safe to combine with OSM node IDs)
        """
        routing_net = RoutingNetwork()
        routing_net.epsg = trails_vector.epsg

        node_id_counter = 0
        line_endpoint_map = {}  # (x, y) -> node_id for snapping

        for i, polyline in enumerate(trails_vector.coordinates):
            # Get line start and end points
            start_pt = polyline[0]
            end_pt = polyline[-1]

            # Snap start point to existing node or create new
            start_node, start_pt_name = _snap_or_create_node(
                routing_net, start_pt, line_endpoint_map,
                snap_distance, node_id_counter
            )
            if start_node == node_id_counter:
                node_id_counter += 1

            # Snap end point to existing node or create new
            end_node, end_pt_name = _snap_or_create_node(
                routing_net, end_pt, line_endpoint_map,
                snap_distance, node_id_counter
            )
            if end_node == node_id_counter:
                node_id_counter += 1

            # Calculate edge weight (Euclidean distance)
            edge_weight = _calculate_polyline_length(polyline)

            # Add bidirectional edge (hiking can go either way)
            routing_net.add_edge(start_node, end_node,
                               weight=edge_weight,
                               length=edge_weight,
                               source='trail',
                               trail_id=i)

        return routing_net

    **Helper function _snap_or_create_node**:
    def _snap_or_create_node(routing_net, point, endpoint_map, snap_distance, next_id):
        """Snap point to existing node or create new node.

        Args:
            routing_net: RoutingNetwork instance
            point: (x, y) tuple for node location
            endpoint_map: Dict mapping coordinates to node_ids
            snap_distance: Squared distance threshold for snapping
            next_id: Next available sequential node ID

        Returns:
            (node_id, snapped_point) tuple
        """
        x, y = point
        snap_dist_sq = snap_distance ** 2

        # Check if point already exists in routing_net (or endpoint_map)
        # Simple distance check (can optimize with KD-tree for large graphs)
        for node_id, existing_pt in routing_net.node_coords.items():
            existing_x, existing_y = existing_pt
            if (x - existing_x)**2 + (y - existing_y)**2 < snap_dist_sq:
                return node_id, existing_pt

        # Create new node
        routing_net.add_node(next_id, x, y)
        return next_id, (x, y)

    **Helper function _calculate_polyline_length**:
    def _calculate_polyline_length(polyline):
        """Calculate total length of polyline segments (Euclidean distance)."""
        total_length = 0.0
        for i in range(len(polyline) - 1):
            x1, y1 = polyline[i]
            x2, y2 = polyline[i+1]
            segment_length = ((x2 - x1)**2 + (y2 - y1)**2)**0.5
            total_length += segment_length
        return total_length
