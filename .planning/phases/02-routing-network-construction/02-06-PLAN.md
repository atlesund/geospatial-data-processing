```python
class RoutingNetwork:
    def __init__(self):
        self.graph = nx.Graph()
        self.node_coords = {}
        self.epsg = None

    def add_node(self, node_id, x, y):
        """Add a georeferenced node."""
        self.graph.add_node(node_id)
        self.node_coords[node_id] = (x, y)

    def add_edge(self, u, v, weight, **attrs):
        """Add a weighted edge between two nodes."""
        self.graph.add_edge(u, v, weight=weight, **attrs)

    def shortest_path(self, source, target):
        """Compute shortest path using Dijkstra."""
        return nx.dijkstra_path(self.graph, source, target, weight='weight')

    def find_nearest_node(self, x, y, k=1):
        """Find k nearest nodes using scipy KD-tree."""
        coords_array = np.array(list(self.node_coords.values()))
        tree = scipy.spatial.KDTree(coords_array)
        dist, idx = tree.query([x, y], k=k)
        node_ids = list(self.node_coords.keys())[idx]
        return node_ids[0], dist
```

# - Terrain mesh: Sequential integers starting from 0 (separate counter)
# Result: No ID collisions - OSM IDs in high range, trails/mesh in low range

    def merge_routing_networks(networks: list[RoutingNetwork]) -> RoutingNetwork:
        """Merge multiple routing networks into unified hybrid graph.

        Combines RoutingNetworks from different data sources (trails, OSM, terrain mesh)
        into a single graph that supports cross-source routing.

        Args:
            networks: List of RoutingNetwork instances to merge

        Returns:
            RoutingNetwork with combined graph and node_coords from all sources

        Raises:
            ValueError: If EPSG codes are inconsistent across networks

        Node ID strategy:
            - OSM: Large integer IDs (e.g., 12345678) - no collision risk
            - Trails: Sequential integers starting from 0 (e.g., 0, 1, 2, ...)
            - Terrain mesh: Sequential integers starting from 0 (separate from trails)
        Result: No ID collisions between sources
        """
        if not networks:
            raise ValueError("At least one network must be provided for merging")

        # Validate EPSG consistency
        epsg_values = set()
        for net in networks:
            if net.epsg is not None:
                epsg_values.add(net.epsg)

        if len(epsg_values) > 1:
            raise ValueError(f"Cannot merge networks with inconsistent EPSG codes: {epsg_values}")

        # Create merged network
        merged = RoutingNetwork()
        if epsg_values:
            merged.epsg = epsg_values.pop()  # Set consistent EPSG
        else:
            merged.epsg = None  # All networks have None EPSG

        # Merge all nodes and edges
        for network in networks:
            for node_id, coords in network.node_coords.items():
                merged.add_node(node_id, coords[0], coords[1])

            for u, v, edge_data in network.graph.edges(data=True):
                weight = edge_data.get('weight', 1.0)
                merged.add_edge(u, v, weight, **edge_data)

        return merged

    **Implementation notes**:
    - Directly copy node_coords and edges using add_node/add_edge methods
    - Use network.graph.edges(data=True) to preserve edge attributes (source='trail', source='osm', source='terrain')
    - Validate EPSG before merging to prevent coordinate system mismatches
    - Raise ValueError if networks list is empty
