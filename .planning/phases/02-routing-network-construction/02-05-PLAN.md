```python
class RoutingNetwork:
    def __init__(self):
        self.graph = nx.Graph()
        self.node_coords = {}
        self.epsg = None

    def add_node(self, node_id, x, y):
        """Add a georeferenced node."""
        self.graph.add_node(node_id)
        self.node_coords[node_id] = (x, y)

    def add_edge(self, u, v, weight, **attrs):
        """Add a weighted edge between two nodes."""
        self.graph.add_edge(u, v, weight=weight, **attrs)
```

    def terrain_mesh_from_raster(raster: Raster, mesh_spacing=100, bbox=None) -> RoutingNetwork:
        """Generate a regular mesh node grid from terrain raster.

        For Phase 2: Creates placeholder mesh structure with equal edge weights.
        Phase 3: Will add terrain-based edge weights using slope/ elevation.

        Args:
            raster: Raster instance with DTM data and world file
            mesh_spacing: Distance between mesh nodes (meters in projection, default 100)
            bbox: Optional bounding box (x_min, y_min, x_max, y_max) - if None, use full raster extent

        Returns:
            RoutingNetwork with regular mesh topology (4-connectivity grid)
        """
        routing_net = RoutingNetwork()
        routing_net.epsg = raster.epsg

        # Get raster extent and pixel size from world file
        world_file = raster._world_file
        pixel_width = world_file[0]
        pixel_height = world_file[3]  # Negative (downward)
        top_left_x = world_file[4]
        top_left_y = world_file[5]

        # Calculate pixel spacing for mesh nodes
        pixel_spacing = mesh_spacing / abs(pixel_width)

        # Determine mesh bounds (use full raster or bbox)
        if bbox:
            x_min, y_min, x_max, y_max = bbox
            # Calculate start/end row/col from bbox
            start_col = int((x_min - top_left_x) / pixel_width)
            end_col = int((x_max - top_left_x) / pixel_width)
            start_row = int((y_min - top_left_y) / pixel_height)
            end_row = int((y_max - top_left_y) / pixel_height)
        else:
            start_row = 0
            end_row = raster.shape[0]
            start_col = 0
            end_col = raster.shape[1]

        # Generate grid of nodes
        node_id_counter = 0

        for row in range(start_row, end_row, int(pixel_spacing)):
            for col in range(start_col, end_col, int(pixel_spacing)):
                # Convert pixel to world coordinates using world file affine transformation
                # x = top_left_x + col * pixel_width + row * rotation_x
                # y = top_left_y + row * pixel_height + col * rotation_y
                # Note: rotation_x and rotation_y are typically 0 for north-up rasters
                x = top_left_x + col * pixel_width + row * world_file[1]
                y = top_left_y + row * pixel_height + col * world_file[2]

                # Add node to routing network
                routing_net.add_node(node_id_counter, x, y)

                # Connect to left neighbor (if not first column)
                if col > start_col:
                    left_id = node_id_counter - int(pixel_spacing)
                    edge_weight = mesh_spacing
                    routing_net.add_edge(node_id_counter, left_id, edge_weight,
                                       length=edge_weight,
                                       source='terrain')

                # Connect to top neighbor (if not first row)
                if row > start_row:
                    top_id = node_id_counter - int((end_col - start_col) / pixel_spacing)
                    edge_weight = mesh_spacing
                    routing_net.add_edge(node_id_counter, top_id, edge_weight,
                                       length=edge_weight,
                                       source='terrain')

                node_id_counter += 1

        return routing_net
